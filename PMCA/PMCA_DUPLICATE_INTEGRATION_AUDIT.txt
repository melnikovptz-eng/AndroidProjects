    39	    public Submission submitShellCommand(String command) {
    40	        if (command == null || command.trim().isEmpty()) {
    41	            return new Submission(false, "", "Пустая команда");
    42	        }
    43	
    44	        File commandFile = new File(bridgeDir, "command.json");
    45	        File tempFile = new File(
    46	                bridgeDir, "command.pmca_" + UUID.randomUUID() + ".tmp");
    47	
    48	        String operationId = UUID.randomUUID().toString();
    49	
    50	        try {
    51	            if (!bridgeDir.isDirectory() && !bridgeDir.mkdirs()) {
    52	                return new Submission(false, operationId,
    53	                        "Папка моста недоступна");
    54	            }
    55	
    56	            // Не затираем команду, которую ещё обрабатывает демон.
    57	            if (commandFile.exists()) {
    58	                PmcaLogger.log(
    59	                        "Termux: предыдущая команда ещё не обработана");
    60	                return new Submission(false, operationId,
    61	                        "Предыдущая команда ещё обрабатывается");
    62	            }
    63	
    64	            JSONObject json = new JSONObject();
    65	            json.put("operation_id", operationId);
    66	            json.put("type", "PMCA");
    67	            json.put("command", "SHELL");
    68	            json.put("data", command);
    69	
    70	            byte[] bytes = json.toString().getBytes(StandardCharsets.UTF_8);
    71	
    72	            try (FileOutputStream out = new FileOutputStream(tempFile, false)) {
    73	                out.write(bytes);
    74	                out.getFD().sync();
    75	            }
    76	
    77	            // Повторная проверка перед передачей команды.
    78	            if (commandFile.exists()) {
    79	                return new Submission(false, operationId,
    80	                        "Мост занят другой командой");
    81	            }
    82	
    83	            if (!tempFile.renameTo(commandFile)) {
    84	                return new Submission(false, operationId,
    85	                        "Не удалось передать команду мосту");
    86	            }
    87	
    88	            PmcaLogger.log("Termux: команда отправлена, ID=" + operationId);
    89	            return new Submission(true, operationId, "Команда передана");
    90	
    91	        } catch (Exception e) {
    92	            PmcaLogger.log("Termux: ошибка отправки: "
    93	                    + e.getClass().getSimpleName());
    94	            return new Submission(false, operationId,
    95	                    "Ошибка отправки: " + e.getClass().getSimpleName());
    96	        } finally {
    97	            // Удаляем только собственный временный файл.
    98	            if (tempFile.exists()) {
    99	                // Оставляем его при ошибке для диагностики.
   100	            }
   101	        }

   104	    public String waitForResult(String operationId, long timeoutMs) {
   105	        if (operationId == null || operationId.isEmpty()) {
   106	            return "ОШИБКА: отсутствует ID команды";
   107	        }
   108	
   109	        File resultFile = new File(bridgeDir, "result.json");
   110	        long deadline = System.currentTimeMillis() + timeoutMs;
   111	
   112	        while (System.currentTimeMillis() < deadline) {
   113	            try {
   114	                if (resultFile.isFile() && resultFile.length() > 0) {
   115	                    byte[] bytes = java.nio.file.Files.readAllBytes(
   116	                            resultFile.toPath());
   117	
   118	                    JSONObject result = new JSONObject(
   119	                            new String(bytes, StandardCharsets.UTF_8));
   120	
   121	                    String resultId = result.optString("operation_id", "");
   122	
   123	                    // Старый ответ не принимается.
   124	                    if (operationId.equals(resultId)) {
   125	                        String status = result.optString("status", "UNKNOWN");
   126	                        String response = result.optString(
   127	                                "response", "Ответ пуст");
   128	
   129	                        PmcaLogger.log(
   130	                                "Termux: получен ответ для ID=" + operationId);
   131	
   132	                        return "Статус: " + status + "\n"
   133	                                + "ID: " + operationId + "\n\n"
   134	                                + response;
   135	                    }
   136	                }
   137	            } catch (Exception ignored) {
   138	                // Демон может ещё записывать JSON. Повторяем чтение.
   139	            }
   140	
   141	            try {
   142	                Thread.sleep(300);
   143	            } catch (InterruptedException e) {
   144	                Thread.currentThread().interrupt();
   145	                return "ОЖИДАНИЕ ПРЕРВАНО";
   146	            }
   147	        }
   148	
   149	        PmcaLogger.log("Termux: тайм-аут ожидания ID=" + operationId);
   150	        return "ТАЙМ-АУТ: ответ пока не получен.\n"
   151	                + "Команда могла выполниться; проверь журнал Termux.\n"
   152	                + "ID: " + operationId;
   153	    }

    25	        if (intent == null) return;
    26	
    27	        Bundle extras = null;
    28	        Bundle result = null;
    29	        String source = "не найден";
    30	
    31	        try {
    32	            extras = intent.getExtras();
    33	        } catch (Exception ignored) {
    34	        }
    35	
    36	        try {

   117	    private static String findString(Bundle bundle, String... names) {
   118	        if (bundle == null) return "";
   119	
   120	        for (String key : bundle.keySet()) {
   121	            if (key == null || key.endsWith("_ORIGINAL_LENGTH")) continue;
   122	
   123	            String normalized = normalize(key);
   124	            boolean match = false;
   125	
   126	            for (String name : names) {
   127	                String expected = normalize(name);
   128	                if (normalized.equals(expected)
   129	                        || normalized.endsWith(expected)) {
   130	                    match = true;
   131	                    break;
   132	                }
   133	            }
   134	
   135	            if (!match) continue;
   136	
   137	            try {
   138	                Object value = bundle.get(key);
   139	                if (value instanceof String) return (String) value;
   140	                if (value instanceof CharSequence) return value.toString();
   141	            } catch (Exception ignored) {
   142	            }
   143	        }
   144	        return "";
   145	    }

   147	    private static int findInt(Bundle bundle, int fallback, String... names) {
   148	        if (bundle == null) return fallback;
   149	
   150	        for (String key : bundle.keySet()) {
   151	            if (key == null) continue;
   152	
   153	            String normalized = normalize(key);
   154	            boolean match = false;
   155	
   156	            for (String name : names) {
   157	                String expected = normalize(name);
   158	                if (normalized.equals(expected)
   159	                        || normalized.endsWith(expected)) {
   160	                    match = true;
   161	                    break;
   162	                }
   163	            }
   164	
   165	            if (!match) continue;
   166	
   167	            try {
   168	                Object value = bundle.get(key);
   169	
   170	                if (value instanceof Number) {
   171	                    return ((Number) value).intValue();
   172	                }
   173	
   174	                if (value instanceof String) {
   175	                    return Integer.parseInt(((String) value).trim());
   176	                }
   177	            } catch (Exception ignored) {
   178	            }
   179	        }
   180	        return fallback;
   181	    }

   183	    private static String normalize(String value) {
   184	        return value.toUpperCase(Locale.ROOT)
   185	                .replace("-", "")
   186	                .replace("_", "")
   187	                .replace(".", "");
   188	    }
