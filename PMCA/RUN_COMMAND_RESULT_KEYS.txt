package com.pmca.bridge.termux;

import android.app.IntentService;
import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Bundle;
import android.os.Parcelable;

import com.pmca.bridge.logging.PmcaLogger;

import java.util.ArrayList;

public final class TermuxResultService extends IntentService {

    public static final String ACTION_RESULT = "com.pmca.bridge.TERMUX_RESULT";

    private static final String RESULT_BUNDLE_KEY =
            "com.termux.app.shell.TermuxService.EXTRA_PLUGIN_RESULT_BUNDLE";

    public TermuxResultService() {
        super("PMCA-TermuxResultService");
    }

    @Override
    protected void onHandleIntent(Intent intent) {
        if (intent == null) return;

        Bundle extras;
        try {
            extras = intent.getExtras();
        } catch (Exception e) {
            extras = null;
        }

        Bundle result = null;
        String source = "не найден";

        // Способ 1: стандартный ключ Termux.
        try {
            result = intent.getBundleExtra(RESULT_BUNDLE_KEY);
            if (result != null) source = "стандартный ключ Termux";
        } catch (Exception ignored) {
        }

        // Способ 2: поиск вложенного Bundle по имени ключа.
        if (result == null && extras != null) {
            for (String key : extras.keySet()) {
                try {
                    Object value = extras.get(key);

                    if (value instanceof Bundle) {
                        Bundle candidate = (Bundle) value;

                        if (containsResultFields(candidate)) {
                            result = candidate;
                            source = "вложенный Bundle: " + key;
                            break;
                        }
                    }

                    // Дополнительная проверка, если результат пришёл как Parcelable.
                    if (value instanceof Parcelable && value instanceof Intent) {
                        Bundle nested = ((Intent) value).getExtras();

                        if (nested != null && containsResultFields(nested)) {
                            result = nested;
                            source = "вложенный Intent: " + key;
                            break;
                        }
                    }
                } catch (Exception ignored) {
                }
            }
        }

        // Способ 3: некоторые варианты ответа могут находиться прямо в extras.
        if (result == null && extras != null && containsResultFields(extras)) {
            result = extras;
            source = "непосредственные extras Intent";
        }

        StringBuilder text = new StringBuilder();

        if (result == null) {
            text.append("ОШИБКА: пакет результата Termux не найден.\n\n")
                .append("Action: ").append(intent.getAction()).append("\n")
                .append("Component: ").append(intent.getComponent()).append("\n")
                .append("Ключи полученного Intent:\n");

            if (extras == null || extras.isEmpty()) {
                text.append("(дополнительных полей нет)");
            } else {
                for (String key : extras.keySet()) {
                    Object value;
                    try {
                        value = extras.get(key);
                    } catch (Exception e) {
                        value = null;
                    }

                    text.append("• ").append(key)
                        .append(" [")
                        .append(value == null ? "null" :
                                value.getClass().getSimpleName())
                        .append("]\n");
                }
            }

            text.append("\nПроверь разрешение RUN_COMMAND, ")
                .append("allow-external-apps и версию Termux.");
        } else {
            String stdout = findString(result, "STDOUT");
            String stderr = findString(result, "STDERR");
            String errmsg = findString(result, "ERRMSG");

            int exitCode = findInt(result, "EXIT_CODE", -999);
            int errorCode = findInt(result, "EXTRA_PLUGIN_RESULT_BUNDLE_ERR", -999);

            text.append("===== РЕЗУЛЬТАТ TERMUX =====\n")
                .append("Источник: ").append(source).append("\n")
                .append("Код завершения: ").append(exitCode).append("\n\n")
                .append("===== STDOUT =====\n")
                .append(stdout.isEmpty() ? "(пусто)" : stdout)
                .append("\n\n===== STDERR =====\n")
                .append(stderr.isEmpty() ? "(пусто)" : stderr);

            if (!errmsg.isEmpty()) {
                text.append("\n\n===== ОШИБКА TERMUX =====\n").append(errmsg);
            }

            if (errorCode != -999) {
                text.append("\n\nВнутренний код Termux: ").append(errorCode);
            }

            if (exitCode == -999) {
                text.append("\n\nПРЕДУПРЕЖДЕНИЕ: поле EXIT_CODE не найдено.");
            }
        }

        String output = text.toString();

        PmcaLogger.log("RUN_COMMAND RESULT: " + source);

        getSharedPreferences("pmca_results", MODE_PRIVATE)
                .edit()
                .putString("last_result", output)
                .apply();

        Intent update = new Intent(ACTION_RESULT);
        update.setPackage(getPackageName());
        update.putExtra("result_text", output);
        sendBroadcast(update);
    }

    private static boolean containsResultFields(Bundle bundle) {
        if (bundle == null) return false;

        for (String key : bundle.keySet()) {
            if (key == null) continue;

            String upper = key.toUpperCase(java.util.Locale.ROOT);

            if (upper.endsWith("STDOUT")
                    || upper.endsWith("STDERR")
                    || upper.endsWith("EXIT_CODE")
                    || upper.endsWith("ERRMSG")) {
                return true;
            }
        }

        return false;
    }

    private static String findString(Bundle bundle, String ending) {
        if (bundle == null) return "";

        for (String key : bundle.keySet()) {
            if (key == null
                    || !key.endsWith(ending)
                    || key.endsWith("_ORIGINAL_LENGTH")) {
                continue;
            }

            try {
                Object value = bundle.get(key);
                if (value instanceof String) return (String) value;
            } catch (Exception ignored) {
            }
        }

        return "";
    }

    private static int findInt(Bundle bundle, String ending, int fallback) {
        if (bundle == null) return fallback;

        for (String key : bundle.keySet()) {
            if (key == null || !key.endsWith(ending)) continue;

            try {
                Object value = bundle.get(key);
                if (value instanceof Integer) return (Integer) value;
            } catch (Exception ignored) {
            }
        }

        return fallback;
    }
}
