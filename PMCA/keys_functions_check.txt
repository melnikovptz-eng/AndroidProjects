# PMCA_COMMAND_COPY_BUTTONS
copy_last_commands() {
    local n="${1:-1}"
    local text

    text="$(fc -ln -"$n" 2>/dev/null)"

    if [ -z "$text" ]; then
        echo "Нет доступных команд"
        return 1
    fi

    # Используем текущий Termux clipboard механизм,
    # но не позволяем ему зависнуть.
    if command -v termux-clipboard-set >/dev/null 2>&1; then
        printf '%s' "$text" | timeout 3 termux-clipboard-set >/dev/null 2>&1
        local rc=$?

        if [ "$rc" -eq 0 ]; then
            echo "Скопировано последних команд: $n"
            return 0
        fi
    fi

    echo "Копирование недоступно"
    return 1
}

1с() { copy_last_commands 1; }
2с() { copy_last_commands 2; }
3с() { copy_last_commands 3; }

# ===== TERMUX HISTORY COPY KEYS =====

copy_last_commands() {
    local n="${1:-1}"
    local start=$((n + 1))
    local text

    text="$(fc -ln -"$start" -2 2>/dev/null)"

    if [ -z "$text" ]; then
        echo "Нет доступных предыдущих команд"
        return 1
    fi

    printf '%s' "$text" | timeout 3 termux-clipboard-set >/dev/null 2>&1

    if [ $? -eq 0 ]; then
        echo "Скопировано команд: $n"
    else
        echo "Буфер Termux:API недоступен"
    fi
}

# ===== PMCA OUTPUT LOGGER =====
export PMCA_OUTPUT_LOG="$HOME/.pmca_output.log"

pmca_output_start() {
    : > "$PMCA_OUTPUT_LOG"
    exec 3>&1 4>&2
    exec > >(tee -a "$PMCA_OUTPUT_LOG") 2>&1
}

pmca_output_stop() {
    exec >&3 2>&4
    exec 3>&- 4>&-
}

# Запускать вывод через PMCA:
pmca_run() {
    pmca_output_start
    "$@"
    rc=$?
    pmca_output_stop
    return $rc
}
# ===== END PMCA OUTPUT LOGGER =====
