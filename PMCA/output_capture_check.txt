===== CAPTURE FUNCTION =====
pmca_output_start () 
{ 
    : > "$PMCA_OUTPUT_LOG";
    exec 3>&1 4>&2;
    exec > >(tee -a "$PMCA_OUTPUT_LOG") 2>&1
}
pmca_output_stop () 
{ 
    exec 1>&3 2>&4;
    exec 3>&- 4>&-
}
pmca_run () 
{ 
    pmca_output_start;
    "$@";
    rc=$?;
    pmca_output_stop;
    return $rc
}
